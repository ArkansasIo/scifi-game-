cmake_minimum_required(VERSION 3.20)
project(spacebattlerpg VERSION 2.1.0 LANGUAGES CXX)

option(SBW_BUILD_TESTS "Build lightweight project tests" ON)
option(SBW_ENABLE_LUA "Enable external Lua integration (requires project-specific Lua wiring)" OFF)
option(SBW_ENABLE_MYSQL "Enable MySQL persistence (requires project-specific MySQL wiring)" OFF)
option(SBW_ENABLE_AUDIO "Enable an external audio backend" OFF)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

file(GLOB_RECURSE SBW_SOURCES CONFIGURE_DEPENDS "src/*.c++")
add_executable(spacebattlerpg ${SBW_SOURCES})
target_include_directories(spacebattlerpg PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")
if(MSVC)
  target_compile_options(spacebattlerpg PRIVATE /W4 /permissive-)
else()
  target_compile_options(spacebattlerpg PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(SBW_ENABLE_LUA)
  target_compile_definitions(spacebattlerpg PRIVATE SBW_ENABLE_LUA=1)
endif()
if(SBW_ENABLE_MYSQL)
  target_compile_definitions(spacebattlerpg PRIVATE SBW_ENABLE_MYSQL=1)
endif()
if(SBW_ENABLE_AUDIO)
  target_compile_definitions(spacebattlerpg PRIVATE SBW_ENABLE_AUDIO=1)
endif()

include(CTest)
if(BUILD_TESTING AND SBW_BUILD_TESTS)
  add_executable(sbw_protocol_tests tests/protocol_tests.c++ src/server/protocol.c++)
  target_include_directories(sbw_protocol_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")
  target_compile_options(sbw_protocol_tests PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-UNDEBUG> $<$<CXX_COMPILER_ID:MSVC>:/UNDEBUG>)
  add_test(NAME protocol_parser COMMAND sbw_protocol_tests)
  add_executable(classic_trek_tests tests/classic_trek_tests.c++ src/game/classic_trek.c++)
  target_include_directories(classic_trek_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")
  target_compile_options(classic_trek_tests PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-UNDEBUG> $<$<CXX_COMPILER_ID:MSVC>:/UNDEBUG>)
  add_test(NAME classic_trek COMMAND classic_trek_tests)
endif()
install(TARGETS spacebattlerpg RUNTIME DESTINATION bin)
